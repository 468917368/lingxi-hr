### 2.3 组件规范

#### 组件结构
```typescript
// 1. 导入顺序：React > 第三方库 > 本地组件 > 工具函数 > 类型定义
import React, { useState, useEffect } from 'react';
import { Button, Card, message } from 'antd';
import { UserOutlined } from '@ant-design/icons';
import JobCard from '@/components/JobCard';
import { getJobList } from '@/services/job';
import type { JobInfo } from '@/types/job';

// 2. 类型定义
interface JobListProps {
  cityCode?: string;
}

// 3. 组件实现
const JobList: React.FC<JobListProps> = ({ cityCode }) => {
  // 3.1 State
  const [jobs, setJobs] = useState<JobInfo[]>([]);
  const [loading, setLoading] = useState(false);

  // 3.2 Hooks
  useEffect(() => {
    fetchJobs();
  }, [cityCode]);

  // 3.3 事件处理函数
  const fetchJobs = async () => {
    setLoading(true);
    try {
      const data = await getJobList({ cityCode });
      setJobs(data);
    } catch (error) {
      message.error('获取岗位列表失败');
    } finally {
      setLoading(false);
    }
  };

  // 3.4 渲染
  return (
    <div className="job-list">
      {jobs.map(job => (
        <JobCard key={job.id} job={job} />
      ))}
    </div>
  );
};

export default JobList;
```

## 4. 状态管理规范

### 4.1 Zustand Store定义

```typescript
// stores/userStore.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

interface UserState {
  // 状态
  userInfo: UserInfo | null;
  token: string | null;
  isLogin: boolean;

  // 操作
  setUserInfo: (user: UserInfo) => void;
  setToken: (token: string) => void;
  logout: () => void;
}

const useUserStore = create<UserState>()(
  persist(
    (set) => ({
      userInfo: null,
      token: null,
      isLogin: false,

      setUserInfo: (user) => set({ userInfo: user, isLogin: true }),
      setToken: (token) => set({ token }),
      logout: () => set({ userInfo: null, token: null, isLogin: false }),
    }),
    {
      name: 'user-storage',
    }
  )
);

export default useUserStore;
```

### 4.2 Store使用规范

- 全局状态使用Zustand Store
- 局部状态使用useState
- 异步操作在Service层处理，不在Store中
- Store保持简洁，只存储状态和同步操作

## 5. API请求规范

### 5.1 请求封装

```typescript
// utils/request.ts
import axios from 'axios';
import { message } from 'antd';
import { tokenManager } from './token';

const request = axios.create({
  baseURL: '/api',
  timeout: 10000,
});

// 请求拦截器
request.interceptors.request.use((config) => {
  const token = tokenManager.getToken();
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// 响应拦截器
request.interceptors.response.use(
  (response) => {
    const { code, message: msg, data } = response.data;
    if (code === 0) {
      return data;
    }
    message.error(msg || '请求失败');
    return Promise.reject(new Error(msg));
  },
  (error) => {
    if (error.response?.status === 401) {
      tokenManager.clearAll();
      window.location.href = '/login';
    }
    message.error('网络异常，请稍后重试');
    return Promise.reject(error);
  }
);

export default request;
```

## 8. SSE流式通信规范

### 8.1 SSE Hook

```typescript
// hooks/useSSE.ts
import { useState, useCallback, useRef } from 'react';

interface SSEOptions {
  url: string;
  onMessage?: (data: any) => void;
  onError?: (error: Error) => void;
  onComplete?: () => void;
}

export function useSSE(options: SSEOptions) {
  const [loading, setLoading] = useState(false);
  const [data, setData] = useState<any>(null);
  const eventSourceRef = useRef<EventSource | null>(null);

  const start = useCallback((params?: Record<string, string>) => {
    setLoading(true);
    setData(null);

    const url = new URL(options.url, window.location.origin);
    if (params) {
      Object.entries(params).forEach(([key, value]) => {
        url.searchParams.append(key, value);
      });
    }

    const eventSource = new EventSource(url.toString());
    eventSourceRef.current = eventSource;

    eventSource.onmessage = (event) => {
      const parsed = JSON.parse(event.data);
      setData(parsed);
      options.onMessage?.(parsed);
    };

    eventSource.onerror = (error) => {
      setLoading(false);
      eventSource.close();
      options.onError?.(new Error('SSE连接失败'));
    };

    return () => {
      eventSource.close();
      setLoading(false);
    };
  }, [options]);

  const stop = useCallback(() => {
    eventSourceRef.current?.close();
    setLoading(false);
  }, []);

  return { loading, data, start, stop };
}
```

## 9. 错误处理规范

### 9.1 全局错误边界

```typescript
// components/ErrorBoundary/index.tsx
import React, { Component, ErrorInfo } from 'react';
import { Result, Button } from 'antd';

interface Props {
  children: React.ReactNode;
}

interface State {
  hasError: boolean;
  error?: Error;
}

class ErrorBoundary extends Component<Props, State> {
  state: State = { hasError: false };

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('ErrorBoundary caught:', error, errorInfo);
    // 上报错误
  }

  render() {
    if (this.state.hasError) {
      return (
        <Result
          status="error"
          title="页面出错了"
          subTitle={this.state.error?.message}
          extra={
            <Button type="primary" onClick={() => window.location.reload()}>
              刷新页面
            </Button>
          }
        />
      );
    }
    return this.props.children;
  }
}

export default ErrorBoundary;
```

### 9.2 错误处理原则

- 网络错误：统一Toast提示
- 401错误：清除Token，跳转登录页
- 403错误：显示无权限页面
- 404错误：显示资源不存在提示
- 500错误：显示服务器错误提示
- 业务错误：根据错误码显示具体提示

## 14. 常见问题FAQ

### Q1: 如何处理Token过期？
A: 在请求拦截器中统一处理，401状态码自动清除Token并跳转登录页。

### Q2: 如何实现SSE流式通信？
A: 使用 `EventSource` API，参考 `useSSE` Hook实现。

### Q3: 如何处理跨域问题？
A: 开发环境使用Umi的proxy配置，生产环境使用Nginx反向代理。
